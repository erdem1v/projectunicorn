. "$(dirname "$0")/env.sh"

out="$S/smoke_$tag"; rm -rf "$out"; mkdir -p "$out"
cases=(); for a in "$@"; do if [ "${a:0:1}" = @ ]; then while IFS= read -r l; do [ -n "$l" ] && cases+=("$l"); done < "${a:1}"; else cases+=("$a"); fi; done
run1() {
  c="$1"

  echo "$(tail -1 "$out/$c.log" | tr -d '\r') [$c]"
}
export -f run1; export tag out tree
printf '%s\n' "${cases[@]}" | xargs -P "${P:-1}" -I{} bash -c 'run1 {}' | tee "$out/summary.txt" | grep -v "SMOKE PASS"
